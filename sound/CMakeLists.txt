# ==============================================================================
# 基础配置：兼容GTest和音频库编译的最低CMake版本要求
# 3.14+ 版本适配GTest的现代CMake接口，同时满足ALSA/Opus等音频库的编译特性
# ==============================================================================
cmake_minimum_required(VERSION 3.14)

# ==============================================================================
# 项目核心信息配置
# - 项目名称：sound_app（音频录制/播放/编码应用）
# - 编程语言：C++（音频处理核心逻辑基于C++17实现）
# - 项目版本：1.0（基础功能版：录音→Opus编码→UDP传输→播放）
# ==============================================================================
project(sound_app    
        LANGUAGES CXX     # 强制使用C++编译器（禁用C编译器）
        VERSION 1.0)      # 项目初始版本号

# ==============================================================================
# C++编译标准与构建模式配置
# ==============================================================================
set(CMAKE_CXX_STANDARD 17)           # 采用C++17标准：适配现代音频库（Opus/ALSA）的接口特性
set(CMAKE_CXX_STANDARD_REQUIRED ON)  # 强制使用C++17，若环境不支持则编译报错（避免隐式降级）
set(CMAKE_BUILD_TYPE Debug)          # 默认Debug模式（便于音频功能调试，发布时改为Release）

# ==============================================================================
# 编译GoogleTest框架（用于音频模块单元测试）
# - ${PROJECT_SOURCE_DIR}：项目根目录（当前CMakeLists.txt所在路径）
# - third_party/googletest：手动下载的GTest源码目录
# - 作用：自动编译gtest/gtest_main库，为后续音频测试用例提供单元测试能力
# ==============================================================================
add_subdirectory(${PROJECT_SOURCE_DIR}/third_party/googletest)

# ==============================================================================
# 生成主程序可执行文件（my_sound）
# 核心功能：音频录制(ALSA)、音频播放(ALSA)、Opus编码/解码、UDP IPC通信
# 源文件说明：
# - src/main.cpp：程序入口（整合所有音频功能流程）
# - src/record.cpp：基于ALSA的音频录制逻辑
# - src/aplay.cpp：基于ALSA的音频播放逻辑
# - src/opus.cpp：Opus音频编码/解码处理
# - src/ipc_udp.cpp：UDP协议的音频数据进程间通信
# ==============================================================================
add_executable(my_sound 
               src/main.cpp
               src/record.cpp
               src/aplay.cpp
               src/opus.cpp
               src/ipc_udp.cpp)

# ==============================================================================
# 指定主程序头文件搜索路径（解决音频模块头文件引用问题）
# - PRIVATE：仅my_sound目标可见，不传递给依赖它的目标（如测试程序）
# - ${PROJECT_SOURCE_DIR}/include：音频模块公共头文件目录（如record.h/aplay.h/opus.h等）
# ==============================================================================
target_include_directories(my_sound
                           PRIVATE 
                           ${PROJECT_SOURCE_DIR}/include)

# ==============================================================================
# 链接音频应用依赖库
# - PRIVATE：库仅链接到my_sound，不传递给依赖目标
# 库说明：
# - asound：ALSA音频驱动库（实现音频硬件的录制/播放）
# - opus：Opus开源音频编码库（低延迟、高压缩比的音频编码）
# - speexdsp：Speex音频处理库（提供音频降噪、回声消除等功能）
# - pthread：POSIX线程库（处理音频采集/播放的异步逻辑）
# ==============================================================================
target_link_libraries(my_sound
                      PRIVATE
                      asound    # ALSA音频核心库（-lasound）
                      opus      # Opus音频编码库（-lopus）
                      speexdsp  # SpeexDSP音频处理库（-lspeexdsp），主要用于降噪
                      pthread)  # 线程库（处理音频异步任务）

# ==============================================================================
# 编译音频模块单元测试程序
# - 进入tests目录执行该目录下的CMakeLists.txt
# - 编译结果：生成测试可执行文件run_tests（包含录音/播放等功能的单元测试）
# ==============================================================================
add_subdirectory(tests)

# ==============================================================================
# 编译&运行说明（新手友好版）
# ==============================================================================
# 1. 构建编译目录（避免源码目录污染）
#    mkdir -p build && cd build && cmake ..
# 2. 编译所有目标（主程序+测试程序）
#    make -j$(nproc)  # -j$(nproc)：利用所有CPU核心加速编译
# 3. 仅编译测试程序
#    make run_tests
# 4. 运行指定测试用例（精准调试音频功能）
#    ./tests/run_tests --gtest_filter=PlayTest.PlayRecordedAudio    # 测试播放录制的音频
#    ./tests/run_tests --gtest_filter=RecordTest.RecordToFileFor5Seconds  # 测试录制5秒音频到文件
