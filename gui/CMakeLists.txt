# ==============================================================================
# 基础配置：兼容GTest和音频库编译的最低CMake版本要求
# 3.14+ 版本适配GTest的现代CMake接口，同时满足ALSA/Opus等音频库的编译特性
# ==============================================================================
cmake_minimum_required(VERSION 3.14)

# ==============================================================================
# 项目核心信息配置
# - 项目名称：sound_app（音频录制/播放/编码应用）
# - 编程语言：C++（音频处理核心逻辑基于C++17实现）
# - 项目版本：1.0（基础功能版：录音→Opus编码→UDP传输→播放）
# ==============================================================================
project(gui_app    
        LANGUAGES CXX     # 强制使用C++编译器（禁用C编译器）
        VERSION 1.0)      # 项目初始版本号

# ==============================================================================
# C++编译标准与构建模式配置
# ==============================================================================
set(CMAKE_CXX_STANDARD 17)           # 采用C++17标准：适配现代音频库（Opus/ALSA）的接口特性
set(CMAKE_CXX_STANDARD_REQUIRED ON)  # 强制使用C++17，若环境不支持则编译报错（避免隐式降级）
set(CMAKE_BUILD_TYPE Debug)          # 默认Debug模式（便于音频功能调试，发布时改为Release）

# ==============================================================================
# 生成主程序可执行文件（my_gui）
# 核心功能：UDP IPC通信
# 源文件说明：
# - src/main.cpp：程序入口
# - src/ipc_udp.cpp：UDP协议的ui数据进程间通信
# ==============================================================================
add_executable(my_gui 
               src/main.cpp
               src/ipc_udp.cpp)

# ==============================================================================
# 指定主程序头文件搜索路径
# - PRIVATE：仅my_gui目标可见，不传递给依赖它的目标（如测试程序）
# - ${PROJECT_SOURCE_DIR}/include：公共头文件目录（如ipc_udp.h等）
# ==============================================================================
target_include_directories(my_gui
                           PRIVATE 
                           ${PROJECT_SOURCE_DIR}/include)

# ==============================================================================
# 链接音频应用依赖库
# - PRIVATE：库仅链接到my_gui，不传递给依赖目标
# 库说明：
# - pthread：POSIX线程库
# ==============================================================================
target_link_libraries(my_gui
                      PRIVATE
                      pthread)

# ==============================================================================
# 编译&运行说明（新手友好版）
# ==============================================================================
# 1. 构建编译目录（避免源码目录污染）
#    mkdir -p build && cd build && cmake ..
# 2. 编译所有目标（主程序+测试程序）
#    make -j$(nproc)  # -j$(nproc)：利用所有CPU核心加速编译
